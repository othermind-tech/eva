#       --python-version 3.11 --generate-hashes
#
# Bump a single package without moving the rest (uv keeps existing pins):
#
#   uv pip compile requirements.txt -o requirements.lock --universal \
#       --python-version 3.11 --generate-hashes --upgrade-package <name>
#
